# ============================================================
# Kopargaon Biodiversity Conservation Portal
# Node.js Package Requirements
# ============================================================
# This project uses Node.js (not Python).
# Install all dependencies with:
#   npm install
#
# Then start the server with:
#   npm start
# or
#   node backend/server.js
#
# The full portal will be available at:
#   http://localhost:3000
# ============================================================

# Node.js Runtime Requirement
node >= 18.0.0
npm >= 9.0.0
